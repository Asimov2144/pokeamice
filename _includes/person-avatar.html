{%- comment -%} The portrait to show for one person on one entry. The photograph taken for that entry comes first (a portrait whose `post` is the entry's file), then the dated portrait nearest the entry's year when it is within five years, then the main avatar. Leaves the path in `pa_src` ("" when the library has no picture). {% include person-avatar.html who=name post=post %} {%- endcomment -%} {%- assign pa_src = "" -%} {%- assign pa_hit = site.data.people | where: "name", include.who | first -%} {%- if pa_hit -%} {%- assign pa_src = pa_hit.avatar | default: "" -%} {%- if pa_hit.portraits and include.post -%} {%- assign pa_year = include.post.date | date: "%Y" | plus: 0 -%} {%- assign pa_gap = 6 -%} {%- for ph in pa_hit.portraits -%} {%- if ph.post and include.post.path contains ph.post -%} {%- assign pa_src = ph.image | default: "" -%} {%- break -%} {%- endif -%} {%- if ph.year -%} {%- assign pa_d = ph.year | minus: pa_year -%} {%- if pa_d < 0 -%}{%- assign pa_d = 0 | minus: pa_d -%}{%- endif -%} {%- if pa_d < pa_gap -%}{%- assign pa_gap = pa_d -%}{%- assign pa_src = ph.image | default: "" -%}{%- endif -%} {%- endif -%} {%- endfor -%} {%- endif -%} {%- endif -%}
